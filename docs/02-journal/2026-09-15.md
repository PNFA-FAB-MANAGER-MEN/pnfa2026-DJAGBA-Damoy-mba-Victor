main.py - TTS hors-ligne sur bouton poussoir, en MicroPython (sans Arduino)
----------------------------------------------------------------------------
Cible : XIAO ESP32S3 flashe avec un firmware MicroPython (pas Arduino).

Principe :
  - Le texte est converti en .wav a l'avance sur le PC (voir generer_audio.py,
    identique a l'etape precedente).
  - Le fichier .wav (mono, 16 bits, 16 kHz) est copie a la racine du systeme
    de fichiers de l'ESP32 (via Thonny, mpremote ou ampy - pas besoin
    d'Arduino ni de LittleFS "upload tool").
  - A l'appui du bouton, ce script lit le fichier et envoie les echantillons
    audio a l'amplificateur via l'objet machine.I2S.

Cablage (exemple, adapte selon ton ampli - ex: MAX98357A) :
    SCK (BCLK) -> GPIO5
    WS  (LRC)  -> GPIO6
    SD  (DIN)  -> GPIO7
    Bouton     -> GPIO2, autre patte a GND (pull-up interne)

Transfert du fichier .wav sur l'ESP32 (depuis un terminal, sur le PC) :
    pip install mpremote
    mpremote connect <PORT_SERIE> cp bienvenue.wav :bienvenue.wav
    mpremote connect <PORT_SERIE> cp main.py :main.py

(remplace <PORT_SERIE> par ex. COM5 sous Windows, /dev/ttyACM0 sous Linux)

from machine import I2S, Pin
import time

# ---------------- Broches ----------------
SCK_PIN = 5
WS_PIN  = 6
SD_PIN  = 7
BOUTON_PIN = 2

# ---------------- Fichier audio ----------------
FICHIER_AUDIO = "bienvenue.wav"

# ---------------- Anti-rebond ----------------
ANTIREBOND_MS = 300
dernier_appui = 0


def lire_entete_wav(f):
    f.seek(0)
    entete = f.read(12)
    if entete[0:4] != b"RIFF" or entete[8:12] != b"WAVE":
        raise ValueError("Fichier WAV invalide")

    frequence = 16000
    bits = 16
    canaux = 1

    while True:
        chunk_id = f.read(4)
        if len(chunk_id) < 4:
            raise ValueError("Chunk 'data' introuvable")
        chunk_taille = int.from_bytes(f.read(4), "little")

        if chunk_id == b"fmt ":
            fmt = f.read(chunk_taille)
            canaux = int.from_bytes(fmt[2:4], "little")
            frequence = int.from_bytes(fmt[4:8], "little")
            bits = int.from_bytes(fmt[14:16], "little")
            if chunk_taille % 2:
                f.seek(1, 1)
        elif chunk_id == b"data":
            return frequence, bits, canaux, chunk_taille
        else:
            f.seek(chunk_taille + (chunk_taille % 2), 1)


def jouer_wav(nom_fichier):
    print("Lecture de", nom_fichier)

    with open(nom_fichier, "rb") as f:
        frequence, bits, canaux, taille_donnees = lire_entete_wav(f)
        audio_format = I2S.MONO if canaux == 1 else I2S.STEREO

        audio_out = I2S(
            0,
            sck=Pin(SCK_PIN),
            ws=Pin(WS_PIN),
            sd=Pin(SD_PIN),
            mode=I2S.TX,
            bits=bits,
            format=audio_format,
            rate=frequence,
            ibuf=20000,
        )

        tampon = bytearray(4096)
        mv = memoryview(tampon)
        restant = taille_donnees
        try:
            while restant > 0:
                taille_lecture = min(len(tampon), restant)
                n = f.readinto(mv[:taille_lecture])
                if n == 0:
                    break
                audio_out.write(mv[:n])
                restant -= n
        finally:
            audio_out.deinit()

    print("Lecture terminee.")
  

def main():
    global dernier_appui

    bouton = Pin(BOUTON_PIN, Pin.IN, Pin.PULL_UP)

    print("Pret (MicroPython, hors-ligne). Appuie sur le bouton pour entendre le message.")

    while True:
        maintenant = time.ticks_ms()
        if bouton.value() == 0 and time.ticks_diff(maintenant, dernier_appui) > ANTIREBOND_MS:
            dernier_appui = maintenant
            jouer_wav(FICHIER_AUDIO)
        time.sleep_ms(20)

import wave
import struct
import os
import subprocess


def _detecter_format(nom_fichier):
    """Renvoie 'RIFF' (wav), 'FORM' (aiff) ou None selon les 4 premiers octets du fichier."""
    with open(nom_fichier, "rb") as f:
        entete = f.read(4)
    if entete == b"RIFF":
        return "RIFF"
    if entete == b"FORM":
        return "FORM"
    return None


def _convertir_aiff_vers_wav(chemin_aiff, chemin_wav):
    """
    Convertit un fichier AIFF en WAV avec afconvert (outil natif macOS).
    Pas de dependance externe : afconvert est fourni avec macOS.
    """
    chemin_temp = chemin_aiff + ".tmp_aiff"
    os.replace(chemin_aiff, chemin_temp)  # on renomme le fichier source pour eviter tout conflit

    resultat = subprocess.run(
        ["afconvert", "-f", "WAVE", "-d", "LEI16", chemin_temp, chemin_wav],
        capture_output=True, text=True
    )

    os.remove(chemin_temp)

    if resultat.returncode != 0:
        raise RuntimeError(f"Echec de la conversion afconvert : {resultat.stderr}")


def _vers_mono(nom_fichier):
    """Si le wav est stereo, le convertit en mono (moyenne des 2 canaux)."""
    with wave.open(nom_fichier, "rb") as wf:
        n_canaux = wf.getnchannels()
        largeur = wf.getsampwidth()
        frequence = wf.getframerate()
        n_frames = wf.getnframes()
        donnees = wf.readframes(n_frames)

    if n_canaux == 1:
        return

    if largeur != 2:
        print("Attention : format audio non standard, conversion mono ignoree.")
        return

    echantillons = struct.unpack("<%dh" % (len(donnees) // 2), donnees)
    mono = [
        (echantillons[i] + echantillons[i + 1]) // 2
        for i in range(0, len(echantillons), 2)
    ]
    donnees_mono = struct.pack("<%dh" % len(mono), *mono)

    with wave.open(nom_fichier, "wb") as wf:
        wf.setnchannels(1)
        wf.setsampwidth(largeur)
        wf.setframerate(frequence)
        wf.writeframes(donnees_mono)


def generer_audio(texte, nom_fichier="Oho.wav", vitesse=160, volume=1.0, voix_index=3):
    chemin_absolu = os.path.abspath(nom_fichier)

    moteur = pyttsx3.init()
    moteur.setProperty('rate', vitesse)
    moteur.setProperty('volume', volume)

    voix = moteur.getProperty('voices')
    if voix_index < len(voix):
        moteur.setProperty('voice', voix[voix_index].id)

    moteur.save_to_file(texte, chemin_absolu)
    moteur.runAndWait()
    moteur.stop()
    del moteur

    if not os.path.exists(chemin_absolu) or os.path.getsize(chemin_absolu) == 0:
        print(f"ERREUR : le fichier n'a pas ete cree ou est vide ({chemin_absolu})")
        return

    # macOS genere en realite de l'AIFF sous l'extension .wav : on corrige si besoin
    format_reel = _detecter_format(chemin_absolu)
    if format_reel == "FORM":
        print("Fichier detecte au format AIFF (typique de macOS), conversion en WAV...")
        _convertir_aiff_vers_wav(chemin_absolu, chemin_absolu)
    elif format_reel != "RIFF":
        print(f"ERREUR : format de fichier non reconnu (en-tete = {format_reel})")
        return

    _vers_mono(chemin_absolu)

    with wave.open(chemin_absolu, "rb") as wf:
        print(f"Fichier genere : {chemin_absolu}")
        print(f"  canaux={wf.getnchannels()}  bits={wf.getsampwidth()*8}  "
              f"frequence={wf.getframerate()} Hz  duree={wf.getnframes()/wf.getframerate():.1f}s")

    taille_ko = os.path.getsize(chemin_absolu) / 1024
    print(f"  taille : {taille_ko:.1f} Ko")


def lister_voix():
    moteur = pyttsx3.init()
    for i, v in enumerate(moteur.getProperty('voices')):
        print(f"{i} - {v.name} ({v.languages})")
    moteur.stop()
    del moteur
