import pyttsx3

def parler(texte, vitesse=160, volume=1.0, voix_index=156):
    """
    Convertit un texte en parole (TTS) hors-ligne.
    - texte : le texte à lire
    - vitesse : mots par minute (par défaut ~170)
    - volume : entre 0.0 et 1.0
    - voix_index : index de la voix (0 = voix par défaut du système)
    """
    moteur = pyttsx3.init()

    # Réglages
    moteur.setProperty('rate', vitesse)
    moteur.setProperty('volume', volume)

    voix = moteur.getProperty('voices')
    if voix_index < len(voix):
        moteur.setProperty('voice', voix[voix_index].id)

    moteur.say(texte)
    moteur.runAndWait()

def lister_voix():
    """Affiche les voix disponibles sur le système."""
    moteur = pyttsx3.init()
    for i, v in enumerate(moteur.getProperty('voices')):
        print(f"{i} - {v.name} ({v.languages})")

def sauvegarder_en_fichier(texte, nom_fichier="sortie.mp3"):
    """Sauvegarde le texte parlé dans un fichier audio."""
    moteur = pyttsx3.init()
    moteur.save_to_file(texte, nom_fichier)
    moteur.runAndWait()

if __name__ == "__main__":
    lister_voix()
    parler(" A, comme, Ananas.")
    sauvegarder_en_fichier("Ce texte sera sauvegardé en fichier audio.", "sortie.mp3")
